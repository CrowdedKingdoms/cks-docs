```graphql
type PlayerWasmModuleVersion {

  versionNo: Int!
  target: PlayerComputeTarget!
  sourceFilesJson: String
  openSource: Boolean!

  compileLog: String
  compiledSizeBytes: BigInt
  createdAt: DateTime!
}
```
